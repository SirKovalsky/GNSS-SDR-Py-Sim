# Зависимости GNSS Sim (GPS L1 C/A + L1C, Galileo E1, SBAS, QZSS).
# Нужен Python 3.10..3.12 для UHD (3.13 работает, но без UHD-передачи).
# UHD 4.10 собран против NumPy 1.x, поэтому на Python < 3.13 держим NumPy < 2.0.
# У NumPy 1.x нет wheel'ов под Python 3.13+, поэтому там ставим NumPy 2.x
# (UHD-биндинги опциональны и всё равно ставятся отдельно).
numpy>=1.24,<2.0; python_version < "3.13"
numpy>=2.0,<3.0; python_version >= "3.13"
PyQt5>=5.15.9

# График спектра (PSD) в GUI и чтение NMEA с u-blox ZED-F9P по COM-порту.
pyqtgraph>=0.13
pyserial>=3.5

# Тесты (scripts/setup_windows.ps1 запускает: python -m pytest tests -q).
pytest>=7.0

# CUDA-ускорение синтеза (необязательно; нужен NVIDIA GPU и драйвер).
# auto-режим сам выберет CUDA, если пакет установлен и GPU доступен:
#   Windows/Linux + CUDA 12.x:  pip install "cupy-cuda12x<14" ^
#       nvidia-cuda-runtime-cu12 nvidia-cuda-nvrtc-cu12 nvidia-nvjitlink-cu12
#   CUDA 11.x:                  pip install "cupy-cuda11x"
# Пакеты nvidia-*-cu12 дают cudart/NVRTC без установки всего CUDA Toolkit:
# gnss_sim сам добавляет их DLL в путь поиска (см. gnss_sim/cuda_env.py).
# Проще всего: scripts/setup_windows.ps1 -Cuda.
# Без CuPy приложение работает на CPU.

# UHD (только для передачи на USRP B210) ставится ОТДЕЛЬНО:
#   Windows: UHD-инсталлятор + pip install uhd==4.10.0.0
#   Linux:   sudo apt install libuhd-dev uhd-host python3-uhd
# Без UHD приложение всё равно работает — генерирует IQ в файл.
